it('Q-010: five concurrent first failures should lock each fresh throttle key', async () => {
  const throttle = new LoginThrottle(db, clock);
  const results = [];
  for (let trial = 0; trial < 10; trial++) {
    const key = throttleKey('username', `qa_parallel_${trial}`);
    await Promise.all(Array.from({ length: 5 }, () => throttle.recordFailure([key])));
    const { rows } = await db.query(
      'SELECT failures, locked_until FROM identity.login_throttle WHERE key=$1',
      [key],
    );
    results.push({ trial, ...rows[0], locked: await throttle.isLocked([key]) });
  }
  console.log('Q010_EVIDENCE', JSON.stringify(results));
  expect(results.every((r) => r.locked)).toBe(true);
});

  const uid = await commands.createAdmin('qa_budget_owner', 'correct horse battery');
  const prices = app.get(PriceService);
  const draft = await prices.createDraft(uid, {
    versionLabel: 'QA',
    note: null,
    items: [
      {
        modelKey: 'qa/model',
        unit: 'input_tokens_per_million',
        priceMicros: 2000000,

      {
        modelKey: 'qa/model',
        unit: 'output_tokens_per_million',

  await app
    .get(BillingAdminService)
    .adjust(uid, uid, {
      direction: 'grant',
      amountMicros: 1000000,
      reason: 'QA only',
      idempotencyKey: 'qa-grant',

  const reservation = await app
    .get(ReservationService)
    .estimateAndReserve({
      account: { kind: 'user', userId: uid },
      purpose: 'chat_reply',
      modelKey: 'qa/model',
      estimate: { inputTokens: 1000, outputTokens: 1000 },
      idempotencyKey: 'qa-reserve',
    });
  expect(reservation.ok).toBe(true);
  const before = await db.query('SELECT reserved_cost_micros FROM billing.platform_daily_budget');
  await app.get(BillingLifecycle).purgeUser(uid);
  const after = await db.query('SELECT reserved_cost_micros FROM billing.platform_daily_budget');
  const holds = await db.query('SELECT count(*) FROM billing.holds');
  console.log(
    'Q011_EVIDENCE',
    JSON.stringify({ before: before.rows, after: after.rows, holds: holds.rows }),
  );
  clock.advance(7200000);
  await app.get(ReservationService).expireHolds();

  expect(Number(final.rows[0].reserved_cost_micros)).toBe(0);
